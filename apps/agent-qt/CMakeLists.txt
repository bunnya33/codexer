cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../desktop/package.json" AGENT_PACKAGE)
string(JSON AGENT_VERSION GET "${AGENT_PACKAGE}" version)
project(CodexerAgent VERSION ${AGENT_VERSION} LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core Gui Widgets Network WebSockets Sql)
include(FetchContent)
set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_COMPRESSION OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_DICTBUILDER OFF CACHE BOOL "" FORCE)
set(ZSTD_LEGACY_SUPPORT OFF CACHE BOOL "" FORCE)
FetchContent_Declare(zstd URL https://codeload.github.com/facebook/zstd/tar.gz/refs/tags/v1.5.7 URL_HASH SHA256=37d7284556b20954e56e1ca85b80226768902e2edabd3b649e9e72c0c9012ee3 SOURCE_SUBDIR build/cmake)
FetchContent_MakeAvailable(zstd)
qt_standard_project_setup()
qt_add_library(CodexerNativeCore STATIC src/native.h src/native_io.cpp src/native_data.cpp src/native_controller.cpp src/native_agent.cpp src/native_usage.cpp src/native_runtime.cpp src/native_operations.cpp src/native_files.cpp src/native_projects.cpp)
target_link_libraries(CodexerNativeCore PUBLIC Qt6::Network Qt6::WebSockets Qt6::Sql libzstd_static)
target_include_directories(CodexerNativeCore PUBLIC src PRIVATE "${zstd_SOURCE_DIR}/lib")
target_compile_definitions(CodexerNativeCore PUBLIC CODEXER_AGENT_VERSION="${AGENT_VERSION}")
qt_add_library(CodexerConnectorUi STATIC src/connector_ui.cpp src/connector_ui.h)
target_link_libraries(CodexerConnectorUi PUBLIC Qt6::Widgets)
target_include_directories(CodexerConnectorUi PUBLIC src)
qt_add_executable(CodexerAgent WIN32 MACOSX_BUNDLE src/main.cpp)
target_link_libraries(CodexerAgent PRIVATE CodexerConnectorUi CodexerNativeCore)
option(CODEXER_NATIVE_TESTS "Build isolated native backend regression tests" ON)
if(CODEXER_NATIVE_TESTS)
  find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Test)
  qt_add_executable(CodexerNativeTests tests/native_tests.cpp)
  target_link_libraries(CodexerNativeTests PRIVATE CodexerNativeCore Qt6::Test)
  qt_add_executable(CodexerUiTests tests/ui_tests.cpp)
  target_link_libraries(CodexerUiTests PRIVATE CodexerConnectorUi Qt6::Test)
  if(MSVC)
    target_compile_options(CodexerNativeTests PRIVATE /utf-8)
    target_compile_options(CodexerUiTests PRIVATE /utf-8)
  endif()
endif()
set_target_properties(CodexerAgent PROPERTIES OUTPUT_NAME "Codexer Agent" MACOSX_BUNDLE_GUI_IDENTIFIER "com.codexer.connector")
if(MSVC)
  target_compile_options(CodexerAgent PRIVATE /utf-8)
  target_compile_options(CodexerConnectorUi PRIVATE /utf-8)
  target_compile_options(CodexerNativeCore PRIVATE /utf-8)
endif()
if(WIN32)
  configure_file(src/version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/version.rc" @ONLY)
  target_sources(CodexerAgent PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
  target_link_libraries(CodexerAgent PRIVATE crypt32 advapi32 bcrypt dwmapi)
elseif(APPLE)
  enable_language(OBJCXX)
  target_sources(CodexerAgent PRIVATE src/platform.mm ../control-native/macos/RelayDeskNative/NativeVault.mm)
  target_compile_options(CodexerAgent PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  target_link_libraries(CodexerAgent PRIVATE "-framework Security" "-framework ServiceManagement" "-framework Foundation")
endif()
